#!/usr/bin/env python3
"""
PC 测试客户端 — TCP 连接 MCU (192.168.1.201:8888)

架构:
  PC(本脚本) ──TCP──► MCU(server) ──USART1──► RK3588

PC 独立心跳线程保持链路存活 + 交互式命令收发.

用法:
  python pc_client.py [host] [port]
  默认: python pc_client.py 192.168.1.201 8888
"""
import socket
import struct
import sys
import time
import threading
import signal
import os

sys.path.insert(0, os.path.join(os.path.dirname(__file__), '..', 'tests'))
try:
    from verify_protocol_v2 import crc16_modbus, crc16_modbus_continue
except ImportError:
    def crc16_modbus(data: bytes) -> int:
        table = []
        for i in range(256):
            crc = i
            for _ in range(8):
                if crc & 1: crc = (crc >> 1) ^ 0xA001
                else: crc >>= 1
            table.append(crc)
        crc = 0xFFFF
        for b in data: crc = (crc >> 8) ^ table[(crc ^ b) & 0xFF]
        return crc

    def crc16_modbus_continue(crc: int, data: bytes) -> int:
        table = []
        for i in range(256):
            c = i
            for _ in range(8):
                if c & 1: c = (c >> 1) ^ 0xA001
                else: c >>= 1
            table.append(c)
        for b in data: crc = (crc >> 8) ^ table[(crc ^ b) & 0xFF]
        return crc


def pack_heartbeat(seq: int) -> bytes:
    """打包心跳帧: [0x5A][Len=4][HCRC][Seq:8B][Type=0x0000][TLV_Len=0][FCRC][0xA5]"""
    payload_len = 4
    total = 16 + payload_len
    out = bytearray(total)
    pos = 0
    out[pos] = 0x5A; pos += 1
    out[pos] = (payload_len >> 8) & 0xFF; pos += 1
    out[pos] = payload_len & 0xFF; pos += 1
    hcrc = crc16_modbus(out[pos-3:pos])
    out[pos] = hcrc & 0xFF; pos += 1
    out[pos] = (hcrc >> 8) & 0xFF; pos += 1
    for i in range(7, -1, -1):
        out[pos] = (seq >> (i*8)) & 0xFF; pos += 1
    out[pos] = 0x00; pos += 1  # Type_H
    out[pos] = 0x00; pos += 1  # Type_L
    out[pos] = 0x00; pos += 1  # TLV_Len_H
    out[pos] = 0x00; pos += 1  # TLV_Len_L
    fcrc = crc16_modbus(bytes(out[1:3]) + bytes(out[13:pos]))
    out[pos] = fcrc & 0xFF; pos += 1
    out[pos] = (fcrc >> 8) & 0xFF; pos += 1
    out[pos] = 0xA5
    return bytes(out)


def pack_cmd(cmd_id: int, params: bytes, seq: int) -> bytes:
    """打包命令帧: [0x5A][Len:2B][HCRC:2B][Seq:8B][Type=0x5000][TLV_Len:2B][CmdID:2B][params][FCRC:2B][0xA5]"""
    tlv_len = 2 + len(params)
    payload_len = 4 + tlv_len
    total = 16 + payload_len
    out = bytearray(total)
    pos = 0
    out[pos] = 0x5A; pos += 1
    out[pos] = (payload_len >> 8) & 0xFF; pos += 1
    out[pos] = payload_len & 0xFF; pos += 1
    hcrc = crc16_modbus(out[pos-3:pos])
    out[pos] = hcrc & 0xFF; pos += 1
    out[pos] = (hcrc >> 8) & 0xFF; pos += 1
    for i in range(7, -1, -1):
        out[pos] = (seq >> (i*8)) & 0xFF; pos += 1
    tlv_start = pos
    out[pos] = 0x50; pos += 1
    out[pos] = 0x00; pos += 1
    out[pos] = (tlv_len >> 8) & 0xFF; pos += 1
    out[pos] = tlv_len & 0xFF; pos += 1
    out[pos] = (cmd_id >> 8) & 0xFF; pos += 1
    out[pos] = cmd_id & 0xFF; pos += 1
    if params:
        out[pos:pos+len(params)] = params; pos += len(params)
    fcrc_data = bytes(out[1:3]) + bytes(out[tlv_start:pos])
    fcrc = crc16_modbus(fcrc_data)
    out[pos] = fcrc & 0xFF; pos += 1
    out[pos] = (fcrc >> 8) & 0xFF; pos += 1
    out[pos] = 0xA5
    return bytes(out)


def parse_frame(data: bytes) -> dict | None:
    """解析一个 TLV 帧"""
    if len(data) < 20 or data[0] != 0x5A or data[-1] != 0xA5:
        return None
    payload_len = (data[1] << 8) | data[2]
    expected = 16 + payload_len
    if len(data) != expected:
        return None
    seq = int.from_bytes(data[5:13], 'big')
    tlv_type = (data[13] << 8) | data[14]
    tlv_len  = (data[15] << 8) | data[16]
    tlv_val  = data[17:17+tlv_len]
    hcrc_ok = crc16_modbus(data[0:3]) == (data[3] | (data[4] << 8))
    fcrc_ok = crc16_modbus(data[1:3] + data[13:13+payload_len]) == \
              (data[13+payload_len] | (data[13+payload_len+1] << 8))
    r = {'seq': seq, 'type': tlv_type, 'tlv_len': tlv_len, 'value': tlv_val,
         'hcrc_ok': hcrc_ok, 'fcrc_ok': fcrc_ok}
    if tlv_type == 0x5000 and tlv_len >= 2:
        r['cmd_id'] = (tlv_val[0] << 8) | tlv_val[1]
        r['params'] = tlv_val[2:] if tlv_len > 2 else b''
        if len(r['params']) >= 1:
            r['status'] = r['params'][0]
    return r


def main():
    host = sys.argv[1] if len(sys.argv) > 1 else '192.168.1.201'
    port = int(sys.argv[2]) if len(sys.argv) > 2 else 8888

    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.settimeout(5)
    try:
        sock.connect((host, port))
        sock.settimeout(0.5)
        print(f'[OK] Connected to {host}:{port}')
    except Exception as e:
        print(f'[FAIL] {e}')
        return 1

    running = True
    seq_lock = threading.Lock()
    pc_seq = 0

    def next_seq():
        nonlocal pc_seq
        with seq_lock:
            s = pc_seq
            pc_seq = (pc_seq + 1) & 0xFFFFFFFFFFFFFFFF
            return s

    def on_exit(sig=None, frame=None):
        nonlocal running
        running = False
    signal.signal(signal.SIGINT, on_exit)
    signal.signal(signal.SIGTERM, on_exit)

    # ── 心跳线程: PC→MCU, 每 1s ──
    def hb_loop():
        while running:
            try:
                seq = next_seq()
                sock.sendall(pack_heartbeat(seq))
            except Exception:
                pass
            time.sleep(1.0)

    threading.Thread(target=hb_loop, daemon=True, name='PC-HB').start()

    # ── 接收线程 ──
    buf = b''
    def recv_loop():
        nonlocal buf, running
        while running:
            try:
                data = sock.recv(1024)
                if not data:
                    print('\n[DISCONNECTED]')
                    running = False
                    break
                buf += data
                while True:
                    idx = buf.find(b'\x5A')
                    if idx < 0: break
                    if idx > 0: buf = buf[idx:]
                    if len(buf) < 16: break
                    plen = (buf[1] << 8) | buf[2]
                    total = 16 + plen
                    if len(buf) < total: break
                    f = parse_frame(buf[:total])
                    if f:
                        flags = ''
                        if not f['hcrc_ok']: flags += ' HCRC!'
                        if not f['fcrc_ok']: flags += ' FCRC!'
                        if f['type'] == 0x0000:
                            print(f'<<< HB    seq=0x{f["seq"]:X}{flags}')
                        elif f['type'] == 0x5000:
                            extra = ''
                            if 'cmd_id' in f:
                                extra = f' CmdID=0x{f["cmd_id"]:04X}'
                                if 'status' in f:
                                    extra += f' status=0x{f["status"]:02X}'
                                    if f['status'] == 0x03:
                                        extra += ' (ACK)'
                                    elif f['status'] == 0x04:
                                        extra += ' (BUSY)'
                            print(f'<<< CMD   seq=0x{f["seq"]:X}{extra}{flags}')
                        else:
                            print(f'<<< 0x{f["type"]:04X} seq=0x{f["seq"]:X}{flags}')
                    buf = buf[total:]
            except socket.timeout:
                continue
            except Exception as e:
                if running:
                    print(f'\n[RECV ERR] {e}')
                break

    threading.Thread(target=recv_loop, daemon=True, name='PC-RECV').start()

    print()
    print('=' * 55)
    print('  Heartbeat: active (1s interval)')
    print('  Commands:')
    print('    <hex> [params]   send command (e.g. 0001 aabb)')
    print('    ?                show CmdID reference')
    print('    q / quit         exit')
    print('=' * 55)
    print()

    try:
        while running:
            try:
                line = input().strip()
                if not line: continue
                if line.lower() in ('q', 'quit', 'exit'): break
                if line == '?':
                    print('  CmdID: 0001-0010=gtest  0012-0014=march  0015-0024=idd')
                    print('         003E=reboot  003F=frame_error')
                    print('  e.g.: 0001 AA BB → CmdID=0x0001, params=0xAA,0xBB')
                    continue

                parts = line.split()
                cmd_id = int(parts[0], 16)
                params = bytes.fromhex(''.join(parts[1:])) if len(parts) > 1 else b''
                seq = next_seq()

                frame = pack_cmd(cmd_id, params, seq)
                sock.sendall(frame)
                print(f'>>> CMD 0x{cmd_id:04X} seq=0x{seq:X} ({len(frame)}B)')

            except (EOFError, KeyboardInterrupt):
                break
            except ValueError as e:
                print(f'  Parse error: {e}')
    finally:
        running = False
        sock.close()

    print('[STOPPED]')
    return 0


if __name__ == '__main__':
    sys.exit(main())
